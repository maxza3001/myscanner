    <script>
        let currentTab = 'scan';
        let batchItems = [];
        let html5QrCode = null;
        let isScanning = true;
        let isTorchOn = false;
        let cameraId = null;
        const DELIMITER = '\n';

        window.addEventListener('load', initApp);

        async function initApp() {
            try {
                const devices = await Html5Qrcode.getCameras();
                if (devices && devices.length > 0) {
                    const backCamera = devices.find(device => device.label.toLowerCase().includes('back') || device.label.toLowerCase().includes('rear'));
                    cameraId = backCamera ? backCamera.id : devices[0].id;
                    startScanner();
                } else {
                    updateStatus("ไม่พบกล้อง", "text-red-600", "bg-red-100");
                }
            } catch (err) {
                updateStatus("ขอสิทธิ์ใช้กล้องไม่สำเร็จ", "text-red-600", "bg-red-100");
            }
        }

        async function startScanner() {
            if (html5QrCode) return;
            html5QrCode = new Html5Qrcode("reader");
            const config = {
                fps: 20,
                qrbox: (w, h) => {
                    const size = Math.floor(Math.min(w, h) * 0.75);
                    return { width: size, height: size };
                }
            };
            try {
                await html5QrCode.start(cameraId, config, onScanSuccess);
                checkTorchSupport();
            } catch (err) { console.error(err); }
        }

        function onScanSuccess(decodedText) {
            if (!isScanning) return;
            if (currentTab === 'scan') {
                playSuccessChime();
                handleSingleScan(decodedText);
            } else if (currentTab === 'batch') {
                handleBatchScan(decodedText);
            }
        }

        function handleSingleScan(text) {
            isScanning = false;
            setTimeout(() => isScanning = true, 2000);
            document.getElementById('scan-result-text').innerText = text;
            document.getElementById('single-result-card').classList.remove('hidden');
            showNotification('สแกนสำเร็จ!', 'bg-blue-600');
        }

        function handleBatchScan(text) {
            if (batchItems.includes(text)) {
                playErrorChime();
                showNotification(`⚠️ รหัสซ้ำ: ${text}`, 'bg-red-500', true);
                isScanning = false;
                setTimeout(() => isScanning = true, 1500);
                return;
            }
            batchItems.push(text);
            updateBatchList();
            playSuccessChime();
            showNotification(`บันทึกรหัสใหม่แล้ว`, 'bg-green-600');
            isScanning = false;
            setTimeout(() => isScanning = true, 1000);
        }

        function switchTab(tab) {
            currentTab = tab;
            document.querySelectorAll('nav button').forEach(btn => btn.className = "flex-1 py-3 font-bold text-gray-400 text-center");
            document.getElementById(`tab-${tab}`).className = "tab-active flex-1 py-3 font-bold text-center";
            document.getElementById('view-scan').style.display = tab === 'scan' ? 'block' : 'none';
            document.getElementById('view-batch').style.display = tab === 'batch' ? 'block' : 'none';
            document.getElementById('view-generate').style.display = tab === 'generate' ? 'block' : 'none';
           
            if (tab === 'generate') {
                document.getElementById('scanner-section').classList.add('hidden');
                if (html5QrCode) html5QrCode.pause();
            } else {
                document.getElementById('scanner-section').classList.remove('hidden');
                if (html5QrCode) html5QrCode.resume();
            }
        }

        function playSuccessChime() {
            playTone(880, 0.1, 0.1, "sine");
            if (navigator.vibrate) navigator.vibrate(40);
        }

        function playErrorChime() {
            const ctx = new (window.AudioContext || window.webkitAudioContext)();
            const osc = ctx.createOscillator();
            const g = ctx.createGain();
            osc.type = "sawtooth";
            osc.frequency.setValueAtTime(150, ctx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(110, ctx.currentTime + 0.4);
            g.gain.linearRampToValueAtTime(0.2, ctx.currentTime + 0.05);
            g.gain.linearRampToValueAtTime(0, ctx.currentTime + 0.4);
            osc.connect(g); g.connect(ctx.destination);
            osc.start(); osc.stop(ctx.currentTime + 0.4);
            if (navigator.vibrate) navigator.vibrate([200, 50, 200]);
        }

        function playTone(freq, duration, volume, type) {
            try {
                const ctx = new (window.AudioContext || window.webkitAudioContext)();
                const osc = ctx.createOscillator();
                const g = ctx.createGain();
                osc.type = type;
                osc.connect(g); g.connect(ctx.destination);
                osc.frequency.value = freq;
                g.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + duration);
                osc.start(); osc.stop(ctx.currentTime + duration);
            } catch(e){}
        }

        function updateStatus(text, textColor, bgColor) {
            const el = document.getElementById('camera-status');
            el.innerText = text;
            el.className = `text-[10px] ${bgColor} ${textColor} px-2 py-1 rounded-full font-bold uppercase`;
        }

        function showNotification(msg, color, shake = false) {
            const el = document.getElementById('notification');
            el.innerText = msg;
            el.className = `fixed top-24 left-1/2 transform -translate-x-1/2 text-white px-8 py-3 rounded-full shadow-2xl z-[100] transition-all duration-300 font-black text-sm ${color}`;
            if (shake) el.classList.add('animate-shake'); else el.classList.remove('animate-shake');
            el.classList.remove('hidden');
            setTimeout(() => el.classList.add('hidden'), 2000);
        }

        function updateBatchList() {
            const list = document.getElementById('batch-list');
            document.getElementById('batch-count').innerText = batchItems.length;
            if (batchItems.length === 0) { list.innerHTML = '<li class="p-8 text-center text-gray-400 text-sm italic">ยังไม่มีรายการ</li>'; return; }
            list.innerHTML = batchItems.map((item, idx) => `<li class="p-4 flex justify-between items-center"><div class="min-w-0 pr-4"><div class="text-[10px] text-gray-400 font-black mb-1">ลำดับ #${idx+1}</div><div class="text-xs font-mono truncate font-bold">${item}</div></div><button onclick="removeBatchItem(${idx})" class="text-red-400 bg-red-50 p-2 rounded-lg"><i class="fas fa-trash-alt"></i></button></li>`).join('');
        }

        function removeBatchItem(idx) { batchItems.splice(idx, 1); updateBatchList(); }
        function clearBatch() { if(confirm('ต้องการล้างข้อมูลสะสมทั้งหมดหรือไม่?')) { batchItems = []; updateBatchList(); } }
       
        function generateMasterCode() {
            if (batchItems.length === 0) return showNotification('ไม่มีข้อมูล', 'bg-red-500');
            switchTab('generate');
            renderQRCode(batchItems.join(DELIMITER));
        }

        function renderQRCode(text) {
            const container = document.getElementById('qrcode-container');
            const barcodeContainer = document.getElementById('barcode-container');
            container.innerHTML = "";
            barcodeContainer.classList.add('hidden');
            document.getElementById('generated-output').classList.remove('hidden');
            document.getElementById('generated-text-display').innerText = text;
            new QRCode(container, { text, width: 220, height: 220, correctLevel : QRCode.CorrectLevel.M });
        }

        async function checkTorchSupport() { try { const state = await html5QrCode.getTorchState(); if (state !== undefined) document.getElementById('torch-button').classList.remove('hidden'); } catch(e){} }
        async function toggleTorch() { isTorchOn = !isTorchOn; await html5QrCode.applyVideoConstraints({ advanced: [{ torch: isTorchOn }] }); }
       
        function copyToClipboard(id) {
            const text = document.getElementById(id).innerText;
            navigator.clipboard.writeText(text).then(() => showNotification('คัดลอกลงคลิปบอร์ดแล้ว', 'bg-gray-800'));
        }

        function shareText(id) {
            const text = document.getElementById(id).innerText;
            if (navigator.share) {
                navigator.share({ title: 'Shared Scan', text: text });
            } else {
                copyToClipboard(id);
            }
        }

        function createManualCode(type) {
            const val = document.getElementById('manual-input').value;
            if (!val) return;
            if (type === 'qr') renderQRCode(val);
            else {
                document.getElementById('qrcode-container').innerHTML = "";
                const bc = document.getElementById('barcode-container');
                bc.classList.remove('hidden');
                document.getElementById('generated-output').classList.remove('hidden');
                document.getElementById('generated-text-display').innerText = val;
                JsBarcode("#barcode-container", val, { width: 2, height: 80, displayValue: false });
            }
        }

        function downloadCode() {
            const canvas = document.querySelector('#qrcode-container canvas');
            const barcodeImg = document.getElementById('barcode-container');
            if(canvas) {
                const link = document.createElement('a');
                link.download = `QR_Code_${Date.now()}.png`;
                link.href = canvas.toDataURL();
                link.click();
            } else if (!barcodeImg.classList.contains('hidden')) {
                showNotification('กรุณาแคปหน้าจอเพื่อบันทึก Barcode', 'bg-gray-700');
            }
        }
